<script>
  const canvas = document.getElementById('canvasStep2');
  const ctx = canvas.getContext('2d');
  const walkInput = document.getElementById('walkPos');
  const walkText = document.getElementById('walk-text');

  function draw() {
    canvas.width = canvas.clientWidth;
    canvas.height = canvas.clientHeight;
    const w = canvas.width;
    const h = canvas.height;
    const steps = parseInt(walkInput.value);

    walkText.innerText = steps === 0 ? '0걸음 (제자리)' : `${Math.abs(steps)}걸음 (${steps < 0 ? '왼쪽' : '오른쪽'})`;

    ctx.clearRect(0, 0, w, h);
    ctx.fillStyle = '#020617';
    ctx.fillRect(0, 0, w, h);

    // 1. 아주 먼 배경 별자리 (상단)
    const margin = 40;
    const numSpacing = (w - margin * 2) / 7;

    ctx.fillStyle = '#1e293b';
    ctx.fillRect(10, 15, w - 20, 45);
    ctx.strokeStyle = '#334155';
    ctx.lineWidth = 1;
    ctx.strokeRect(10, 15, w - 20, 45);

    for (let i = 0; i < 8; i++) {
      const x = margin + i * numSpacing;
      ctx.fillStyle = '#64748b';
      ctx.beginPath();
      ctx.arc(x, 26, 2, 0, Math.PI * 2);
      ctx.fill();

      ctx.fillStyle = '#cbd5e1';
      ctx.font = 'bold 14px Pretendard, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText(`배경 ${i + 1}`, x, 48);
    }

    // 2. 별과 고정된 배경 상의 빨간 점 위치
    const starX = w / 2;
    const starY = 110;

    // 빨간 점: 배경 판 중앙(배경 4~5 사이)에 고정
    const redDotX = w / 2;
    const redDotY = 37;

    // 3. 별 ↔ 고정된 빨간 점을 연결하는 노란 점선
    ctx.strokeStyle = '#facc15';
    ctx.lineWidth = 2;
    ctx.setLineDash([4, 4]);
    ctx.beginPath();
    ctx.moveTo(starX, starY);
    ctx.lineTo(redDotX, redDotY);
    ctx.stroke();
    ctx.setLineDash([]);

    // 배경 상의 고정된 빨간 점
    ctx.fillStyle = '#ef4444';
    ctx.beginPath();
    ctx.arc(redDotX, redDotY, 5, 0, Math.PI * 2);
    ctx.fill();

    // 별 (노란 동그라미)
    ctx.fillStyle = '#facc15';
    ctx.beginPath();
    ctx.arc(starX, starY, 10, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowColor = '#facc15';
    ctx.shadowBlur = 10;
    ctx.fill();
    ctx.shadowBlur = 0;

    ctx.fillStyle = '#fef08a';
    ctx.font = '12px Pretendard, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('⭐ 별', starX, starY + 22);

    // 4. 관찰자 (사람) - 걸음에 따라 양옆으로 이동
    const observerX = (w / 2) + steps * (w * 0.025);
    const observerY = h - 35;

    ctx.fillStyle = '#38bdf8';
    ctx.beginPath();
    ctx.arc(observerX, observerY - 10, 9, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillRect(observerX - 5, observerY - 1, 10, 16);

    ctx.fillStyle = '#94a3b8';
    ctx.font = '12px Pretendard, sans-serif';
    ctx.fillText('🚶 사람', observerX, observerY + 28);
  }

  walkInput.addEventListener('input', draw);

  // OX 퀴즈 처리
  let selectedOXVal = null;
  function selectOX(ans) {
    selectedOXVal = ans;
    const btns = document.querySelectorAll('.ox-btn');
    btns.forEach(b => b.classList.remove('selected'));

    if (ans === 'O') btns[0].classList.add('selected');
    else btns[1].classList.add('selected');

    const fb = document.getElementById('ox-feedback');
    if (ans === 'O') {
      fb.className = 'feedback correct';
      fb.innerText = '🎉 정답입니다! (O) 별은 지구와 엄청나게 멀리 떨어져 있어서 사람이 몇 걸음 걸어가는 정도로는 시선 방향의 변화를 전혀 느낄 수 없습니다.';
    } else {
      fb.className = 'feedback incorrect';
      fb.innerText = '❌ 다시 생각해 보세요. 몇 걸음 걸어갈 때 별의 위치가 달라져 보였나요? 별은 너무 멀리 있어서 위치 변화가 보이지 않습니다.';
    }
  }

  // 최종 정리 채점
  function checkFinalSummary() {
    const val = document.getElementById('blankChoice').value;
    const fb = document.getElementById('summary-feedback');

    if (!val) {
      fb.className = 'feedback incorrect';
      fb.innerText = '⚠️ 빈칸에 들어갈 단어를 선택해 주세요!';
      return;
    }

    if (val === '많이') {
      fb.className = 'feedback correct';
      fb.innerText = '🎉 완벽하게 정리하셨습니다! 사람이 몇 걸음 걸어가는 정도가 아니라, 지구가 태양 주위를 도는 것처럼 "엄청나게 먼 거리"를 이동해야 비로소 별의 위치 변화가 보입니다.';
    } else {
      fb.className = 'feedback incorrect';
      fb.innerText = '❌ 다시 선택해 보세요. 조금 이동했을 때는 별의 위치 변화가 전혀 관찰되지 않았습니다.';
    }
  }

  window.addEventListener('resize', draw);
  draw();
</script>

<script>
  // 버튼 잠금 해제 함수
  function unlockNextStep() {
    const nextBtn = document.getElementById('nextBtn');
    if (nextBtn) {
      nextBtn.disabled = false;
      nextBtn.style.opacity = '1';
      nextBtn.style.cursor = 'pointer';
      nextBtn.style.backgroundColor = '#38bdf8';
      nextBtn.style.color = '#0f172a';
      nextBtn.innerText = 'STEP 3: 지구 공전과 연주시차 이해하기 ➔';
    }
  }

  // 페이지 이동 함수
  function goToNext() {
    const nextBtn = document.getElementById('nextBtn');
    if (nextBtn && !nextBtn.disabled) {
      location.href = 'step3.html';
    }
  }

  // 피드백 감지 및 잠금 해제 스크립트
  const observer2 = new MutationObserver(function(mutations) {
    mutations.forEach(function(mutation) {
      const el = mutation.target;
      const text = el.innerText || el.textContent || '';
      if (el.classList.contains('correct') || text.includes('정답') || text.includes('완료')) {
        unlockNextStep();
      }
    });
  });

  document.addEventListener('DOMContentLoaded', function() {
    // 모든 feedback 클래스 및 feedback ID를 감시
    const targets = document.querySelectorAll('.feedback, [id*="feedback"], [class*="feedback"]');
    targets.forEach(function(el) {
      observer2.observe(el, { attributes: true, childList: true, characterData: true, subtree: true });
    });
  });
</script>
